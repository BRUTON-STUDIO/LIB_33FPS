local function RegisterInstance(TargetInstance, InstanceType)
	table.insert(Library.Instances, {
		Instance = TargetInstance,
		Type = InstanceType,
	})
	return TargetInstance
end

local function SetParents(ParentElement, ChildrenTable)
	if ChildrenTable then
		table.foreach(ChildrenTable, function(_, Child)
			Child.Parent = ParentElement
		end)
	end
	return ParentElement
end

local function SetProperties(TargetObject, PropertiesTable)
	if PropertiesTable then
		table.foreach(PropertiesTable, function(Property, Value)
			TargetObject[Property] = Value
		end)
	end
	return TargetObject
end

local function CreateInstance(...)
	local Arguments = {...}
	if type(Arguments) == "table" then
		local NewInstance = Instance.new(Arguments[1])
		if type(Arguments[2]) ~= "table" then
			if typeof(Arguments[2]) == "Instance" then
				NewInstance.Parent = Arguments[2]
				SetProperties(NewInstance, Arguments[3])
				SetParents(NewInstance, Arguments[4])
			end
		else
			SetProperties(NewInstance, Arguments[2])
			SetParents(NewInstance, Arguments[3])
		end
		return NewInstance
	end
end

local function LoadSaveFile(FileName)
	if readfile and (isfile and isfile(FileName)) then
		local Decoded = HttpService:JSONDecode(readfile(FileName))
		if type(Decoded) == "table" then
			if rawget(Decoded, "UISize") then
				Library.Save.UISize = Decoded.UISize
			end
			if rawget(Decoded, "TabSize") then
				Library.Save.TabSize = Decoded.TabSize
			end
			if rawget(Decoded, "Theme") and VerifyTheme(Decoded.Theme) then
				Library.Save.Theme = Decoded.Theme
			end
		end
	end
end

local EventUtils = {
	InsertCallback = function(_, CallbacksTable, Func)
		if type(Func) == "function" then
			table.insert(CallbacksTable, Func)
		end
		return Func
	end,
	FireCallback = function(_, CallbacksTable, ...)
		for _, Func in ipairs(CallbacksTable) do
			if type(Func) == "function" then
				task.spawn(Func, ...)
			end
		end
	end,
	ToggleVisible = function(_, Target, Visible)
		if Visible == nil then
			Visible = Target.Visible
		end
		Target.Visible = Visible
	end,
	ToggleParent = function(_, Target, Bool, Parent)
		if Bool == nil then
			if Target.Parent then
				Parent = false
			end
			Target.Parent = Parent
		else
			Target.Parent = Bool
		end
	end,
	GetConnectionFunctions = function(_, ConnectionList, Func)
		return {
			Function = Func,
			Connected = true,
			Disconnect = function(Self)
				if Self.Connected then
					table.remove(ConnectionList, table.find(ConnectionList, Self.Function))
					Self.Connected = false
				end
			end,
			Fire = function(Self, ...)
				if Self.Connected then
					task.spawn(Self.Function, ...)
				end
			end,
		}
	end,
	GetCallback = function(_, Data, Index)
		local TargetCB = Data[Index] or Data.Callback or function() end
		return type(TargetCB) ~= "table" and { TargetCB } or {
			function(Value)
				TargetCB[1][TargetCB[2]] = Value
			end,
		}
	end,
}

local Registry = {}
local Connection = Library.Connection

local function RegisterEvents(EventsList)
	if type(EventsList) == "table" then
		for _, EventName in ipairs(EventsList) do
			local EventTable = {}
			local EventObj = {}

			Connection[EventName] = EventObj
			Registry[EventName] = EventTable
			EventObj.Name = EventName

			function EventObj.Connect(_, Func)
				if type(Func) == "function" then
					table.insert(EventTable, Func)
				end
			end

			function EventObj.Once(_, Func)
				if type(Func) == "function" then
					local ConnectionRef = nil
					ConnectionRef = EventUtils:GetConnectionFunctions(EventTable, function(...)
						task.spawn(Func, ...)
						ConnectionRef:Disconnect()
					end)
					return ConnectionRef
				end
			end
		end
	end
end

function Connection.FireConnection(_, Event, ...)
	local EventList = type(Event) == "string" and Registry[Event] or Registry[Event.Name]
	for _, Func in pairs(EventList) do
		task.spawn(Func, ...)
	end
end

local function IsFlagSaved(FlagName)
	return type(FlagName) == "string" and Flags[FlagName] ~= nil
end

local function GetFlagValue(FlagName)
	if type(FlagName) ~= "string" then
		return false
	end
	return Flags[FlagName]
end

local function SetFlagValue(FlagName, Value)
	if FlagName and (Value ~= Flags[FlagName] or type(Value) == "table") then
		Flags[FlagName] = Value
		Connection:FireConnection("FlagsChanged", FlagName, Value)
	end
end

local IsSaving = nil

Connection.FlagsChanged:Connect(function(_, _)
	local ScriptFile = Settings.ScriptFile
	if not IsSaving and (ScriptFile and writefile) then
		IsSaving = true
		task.wait(0.1)
		IsSaving = false

		local Success, Encoded = pcall(function()
			return HttpService:JSONDecode(Flags)
		end)

		if Success and pcall(writefile, ScriptFile, Encoded) then
			Connection:FireConnection("FileSaved", "Script-Flags", ScriptFile, Encoded)
		end
	end
end)

local function RegisterClickEvent(Button, Callback)
	Button.InputBegan:Connect(function(Input)
		if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
			while UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) do
				task.wait()
			end
		end
		Callback()
	end)

local function PlayTween(Config)
	local Target = Config[1] or Config.Instance
	local Property = Config[2] or Config.Prop
	local Value = Config[3] or Config.NewVal
	local Time = Config[4] or (Config.Time or 0.5)
	local Wait = Config[5] or (Config.wait or false)

	local Tween = TweenService:Create(Target, TweenInfo.new(Time, Enum.EasingStyle.Quint), {
		[Property] = Value,
	})
	Tween:Play()
	if Wait then
		Tween.Completed:Wait()
	end
	return Tween
end

local function MakeDraggable(Frame)
	task.spawn(function()
		SetProperties(Frame, {
			Active = true,
			AutoButtonColor = false,
		})

		local DragStart = nil
		local StartPos = nil
		local IsDragging = nil

		local function UpdateDrag(Input)
			local Delta = Input.Position - DragStart
			PlayTween({
				Frame,
				"Position",
				UDim2.new(StartPos.X.Scale, StartPos.X.Offset + Delta.X / ScaleFactor, StartPos.Y.Scale, StartPos.Y.Offset + Delta.Y / ScaleFactor),
				0.35,
			})
		end

		Frame.MouseButton1Down:Connect(function()
			IsDragging = true
		end)

		Frame.InputBegan:Connect(function(Input)
			if Input.UserInputType == Enum.UserInputType.MouseButton1 or Input.UserInputType == Enum.UserInputType.Touch then
				StartPos = Frame.Position
				DragStart = Input.Position

				while UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) do
					RunService.Heartbeat:Wait()
					if IsDragging then
						UpdateDrag(Input)
					end
				end
				IsDragging = false
			end
		end)
	end)
	return Frame
end

local function GetPropertyForType(InstanceObj)
	if InstanceObj:IsA("Frame") then return "BackgroundColor3"
	elseif InstanceObj:IsA("ImageLabel") then return "ImageColor3"
	elseif InstanceObj:IsA("TextLabel") then return "TextColor3"
	elseif InstanceObj:IsA("ScrollingFrame") then return "ScrollBarImageColor3"
	elseif InstanceObj:IsA("UIStroke") then return "Color"
	end
	return ""
end

function Library.GetIcon(_, IconName)
	if IconName:find("rbxassetid://") or IconName:len() < 1 then
		return IconName
	end

	local ProcessedName = IconName:lower():gsub("lucide", ""):gsub("-", "")
	for Name, Id in pairs(Library.Icons) do
		if Name:gsub("lucide", ""):gsub("-", "") == ProcessedName then
			return Id
		end
	end

	for Name, Id in pairs(Library.Icons) do
		if Name:gsub("lucide", ""):gsub("-", ""):find(ProcessedName) then
			return Id
		end
	end

	return ProcessedName
end

function Library.SetTheme(_, ThemeName)
	if VerifyTheme(ThemeName) then
		Library.Save.Theme = ThemeName
		SaveToFile("bruton hub with bruton library.json", Library.Save)
		ActiveTheme = Library.Themes[ThemeName]

		Connection:FireConnection("ThemeChanged", ThemeName)
		table.foreach(Library.Instances, function(_, Item)
			if Item.Type == "Gradient" then
				Item.Instance.Color = ActiveTheme["Color Hub 1"]
			elseif Item.Type == "Frame" then
				Item.Instance.BackgroundColor3 = ActiveTheme["Color Hub 2"]
			elseif Item.Type == "Stroke" then
				Item.Instance[GetPropertyForType(Item.Instance)] = ActiveTheme["Color Stroke"]
			elseif Item.Type == "Theme" then
				Item.Instance[GetPropertyForType(Item.Instance)] = ActiveTheme["Color Theme"]
			elseif Item.Type == "Text" then
				Item.Instance[GetPropertyForType(Item.Instance)] = ActiveTheme["Color Text"]
			elseif Item.Type == "DarkText" then
				Item.Instance[GetPropertyForType(Item.Instance)] = ActiveTheme["Color Dark Text"]
			elseif Item.Type == "ScrollBar" then
				Item.Instance[GetPropertyForType(Item.Instance)] = ActiveTheme["Color Theme"]
			end
		end)
	end
end

		function TabFunctions.AddToggle(_, Config)
			local ToggleTitle = Config[1] or Config.Name or (Config.Title or "Toggle")
			local ToggleDesc = Config.Desc or (Config.Description or "")
			local Callbacks = EventUtils:GetCallback(Config, 3)
			local ToggleFlag = Config[4] or (Config.Flag or false)
			local DefaultState = Config[2] or (Config.Default or false)

			if IsFlagSaved(ToggleFlag) then
				DefaultState = GetFlagValue(ToggleFlag)
			end

			local ToggleInstance, BaseController = CreateOptionBase(TabContainer, ToggleTitle, ToggleDesc, UDim2.new(1, -38))
			local SwitchBg = RegisterInstance(
				CreateInstance("Frame", ToggleInstance, {
					Size = UDim2.new(0, 35, 0, 18),
					Position = UDim2.new(1, -10, 0.5),
					AnchorPoint = Vector2.new(1, 0.5),
					BackgroundColor3 = ActiveTheme["Color Stroke"],
				}),
				"Stroke"
			)
			CreateElement("Corner", SwitchBg, UDim.new(0.5, 0))

			local SwitchKnob = RegisterInstance(
				CreateInstance(
					"Frame",
					CreateInstance("Frame", SwitchBg, {
						BackgroundTransparency = 1,
						Size = UDim2.new(0.8, 0, 0.8, 0),
						Position = UDim2.new(0.5, 0, 0.5, 0),
						AnchorPoint = Vector2.new(0.5, 0.5),
					}),
					{
						Size = UDim2.new(0, 12, 0, 12),
						Position = UDim2.new(0, 0, 0.5),
						AnchorPoint = Vector2.new(0, 0.5),
						BackgroundColor3 = ActiveTheme["Color Theme"],
					}
				),
				"Theme"
			)
			CreateElement("Corner", SwitchKnob, UDim.new(0.5, 0))

			local IsUpdating = nil

			local function UpdateState(NewState)
				if not IsUpdating then
					DefaultState = NewState
					IsUpdating = true

					SetFlagValue(ToggleFlag, DefaultState)
					EventUtils:FireCallback(Callbacks, DefaultState)

					if DefaultState then
						PlayTween({ SwitchKnob, "Position", UDim2.new(1, 0, 0.5), 0.25 })
						PlayTween({ SwitchKnob, "BackgroundTransparency", 0, 0.25 })
						PlayTween({ SwitchKnob, "AnchorPoint", Vector2.new(1, 0.5), 0.25 })
					else
						PlayTween({ SwitchKnob, "Position", UDim2.new(0, 0, 0.5), 0.25 })
						PlayTween({ SwitchKnob, "BackgroundTransparency", 0.8, 0.25 })
						PlayTween({ SwitchKnob, "AnchorPoint", Vector2.new(0, 0.5), 0.25 })
					end

					IsUpdating = false
				end
			end

			task.spawn(UpdateState, DefaultState)
			ToggleInstance.Activated:Connect(function()
				UpdateState(not DefaultState)
			end)

			return {
				Visible = function(_, ...) EventUtils:ToggleVisible(ToggleInstance, ...) end,
				Destroy = function(_) ToggleInstance:Destroy() end,
				Callback = function(_, ...) EventUtils:InsertCallback(Callbacks, ...)() end,
				Set = function(_, Value, Extra)
					if type(Value) ~= "string" or type(Extra) ~= "string" then
						if type(Value) ~= "string" then
							if type(Value) ~= "boolean" then
								if type(Value) == "function" then Callbacks = Value end
							else
								if IsUpdating and Extra then
									repeat task.wait() until not IsUpdating
								end
								task.spawn(UpdateState, Value)
							end
						else
							BaseController:SetTitle(Value)
						end
					else
						BaseController:SetTitle(Value)
						BaseController:SetDesc(Extra)
					end
				end,
			}
		end

			local DropTitle = Config[1] or Config.Name or (Config.Title or "Dropdown")

			local DisplayLabel = RegisterInstance(

			local ArrowIcon = CreateInstance("ImageLabel", DropButtonBg, {
				Size = UDim2.new(0, 15, 0, 15),

				AnchorPoint = Vector2.new(1, 0.5),

			})

			local OverlayButton = CreateInstance("TextButton", DropdownFolder, {
				Name = "AntiClick",
				Size = UDim2.new(1, 0, 1, 0),
				BackgroundTransparency = 1,
				Visible = false,
				Text = "",
			})

			local DropdownListFrame = CreateInstance("Frame", OverlayButton, {
				Size = UDim2.new(DropButtonBg.Size.X, 0, 0),
				BackgroundTransparency = 0.1,
				BackgroundColor3 = Color3.fromRGB(255, 255, 255),

				Name = "DropdownFrame",
				ClipsDescendants = true,
				Active = true,
			})

			CreateElement("Corner", DropdownListFrame)
			CreateElement("Stroke", DropdownListFrame)
			CreateElement("Gradient", DropdownListFrame, { Rotation = 60 })

			local ListScroll = RegisterInstance(
				CreateInstance("ScrollingFrame", DropdownListFrame, {
					ScrollBarImageColor3 = ActiveTheme["Color Theme"],
					Size = UDim2.new(1, 0, 1, 0),
					ScrollBarThickness = 1.5,
					BackgroundTransparency = 1,
					BorderSizePixel = 0,
					CanvasSize = UDim2.new(),
					ScrollingDirection = "Y",
					AutomaticCanvasSize = "Y",
					Active = true,
				}, {
					CreateInstance("UIPadding", {
						PaddingLeft = UDim.new(0, 8),
						PaddingRight = UDim.new(0, 8),
						PaddingTop = UDim.new(0, 5),
						PaddingBottom = UDim.new(0, 5),
					}),
					CreateInstance("UIListLayout", { Padding = UDim.new(0, 4) }),
				}),
				"ScrollBar"
			)

			local CalculatedHeight = 5
			local IsDropdownBusy = nil

			local function CloseDropdown()
				IsDropdownBusy = true
				PlayTween({ ArrowIcon, "Rotation", 0, 0.2 })
				PlayTween({ DropdownListFrame, "Size", UDim2.new(0, 152, 0, 0), 0.2, true })
				PlayTween({ ArrowIcon, "ImageColor3", Color3.fromRGB(255, 255, 255), 0.2 })
				ArrowIcon.Image = "rbxassetid://10709791523"
				OverlayButton.Visible = false
				IsDropdownBusy = false
			end

			local function GetCurrentHeight()
				return UDim2.fromOffset(152, CalculatedHeight)
			end

			local function RecalculateHeight()
				local ItemCount = 0
				for _, Child in pairs(ListScroll:GetChildren()) do
					if Child:IsA("Frame") or Child.Name == "Option" then
						ItemCount = ItemCount + 1
					end
				end

				CalculatedHeight = math.clamp(ItemCount, 0, 10) * 25 + 10
				if OverlayButton.Visible then
					OverlayButton.Visible = true
					PlayTween({ DropdownListFrame, "Size", GetCurrentHeight(), 0.2, true })
				end
			end

			local function ToggleDropdown()
				if not IsDropdownBusy then
					IsDropdownBusy = true
					if OverlayButton.Visible then
						ArrowIcon.Image = "rbxassetid://10709791523"
						PlayTween({ ArrowIcon, "ImageColor3", Color3.fromRGB(255, 255, 255), 0.2 })
						PlayTween({ DropdownListFrame, "Size", UDim2.new(0, 152, 0, 0), 0.2, true })
						OverlayButton.Visible = false
					else
						OverlayButton.Visible = true
						ArrowIcon.Image = "rbxassetid://10709790948"
						PlayTween({ ArrowIcon, "ImageColor3", ActiveTheme["Color Theme"], 0.2 })
						PlayTween({ DropdownListFrame, "Size", GetCurrentHeight(), 0.2, true })
					end
					IsDropdownBusy = false
				end
			end

			local function AlignDropdownPosition()
				local ScreenSize = MainScreenGui.AbsoluteSize
				local ButtonPos = DropButtonBg.AbsolutePosition
				local PosX = math.clamp(ButtonPos.X / ScaleFactor, 0, ScreenSize.X / ScaleFactor - DropdownListFrame.Size.X.Offset)
				local PosY = math.clamp(ButtonPos.Y / ScaleFactor, 0, ScreenSize.Y / ScaleFactor)
				local AnchorY = ButtonPos.Y > ScreenSize.Y / 1.4 and 1 or (CalculatedHeight > 80 and 0.5 or 0)

				DropdownListFrame.AnchorPoint = Vector2.new(0, AnchorY)
				PlayTween({ DropdownListFrame, "Position", UDim2.fromOffset(PosX, PosY), 0.1 })
			end

			local RawDefault = type(DropDefault) ~= "table" and { DropDefault } or DropDefault
			local OptionNodes = {}
			local SelectedValue = IsMultiSelect and {} or (IsFlagSaved(DropFlag) and GetFlagValue(DropFlag) or RawDefault[1])

			if IsMultiSelect then
				if IsFlagSaved(DropFlag) then
					RawDefault = GetFlagValue(DropFlag) or RawDefault
				end

				for Key, Val in pairs(RawDefault) do
					if type(Key) ~= "string" or not (DropOptions[Key] or table.find(DropOptions, Key)) then
						if DropOptions[Val] then
							SelectedValue[Val] = true
						end
					else
						SelectedValue[Key] = Val
					end
				end
			end

			local function FireDropdownCallback()
				SetFlagValue(DropFlag, IsMultiSelect and SelectedValue or tostring(SelectedValue))
				EventUtils:FireCallback(Callbacks, SelectedValue)
			end

			local function UpdateDisplayText()
				if IsMultiSelect then
					local SelectedNames = {}
					for Name, Enabled in pairs(SelectedValue) do
						if Enabled then
							table.insert(SelectedNames, Name)
						end
					end
					DisplayLabel.Text = #SelectedNames > 0 and table.concat(SelectedNames, ", ") or "..."
				else
					DisplayLabel.Text = tostring(SelectedValue or "...")
				end
			end

			local function RefreshVisualNodes()
				if IsMultiSelect then
					for _, NodeData in pairs(OptionNodes) do
						local Nodes = NodeData.nodes
						local IsActive = NodeData.Stats
						PlayTween({ Nodes[2], "BackgroundTransparency", IsActive and 0 or 0.8, 0.35 })
						PlayTween({ Nodes[2], "Size", IsActive and UDim2.fromOffset(4, 12) or UDim2.fromOffset(4, 4), 0.35 })
						PlayTween({ Nodes[3], "TextTransparency", IsActive and 0 or 0.4, 0.35 })
					end
				else
					for _, NodeData in pairs(OptionNodes) do
						local IsSelected = NodeData.Value == SelectedValue
						local Nodes = NodeData.nodes
						PlayTween({ Nodes[2], "BackgroundTransparency", IsSelected and 0 or 1, 0.35 })
						PlayTween({ Nodes[2], "Size", IsSelected and UDim2.fromOffset(4, 14) or UDim2.fromOffset(4, 4), 0.35 })
						PlayTween({ Nodes[3], "TextTransparency", IsSelected and 0 or 0.4, 0.35 })
					end
				end
				UpdateDisplayText()
			end

			local function SelectNode(NodeData)
				if IsMultiSelect then
					NodeData.Stats = not NodeData.Stats
					NodeData.LastCB = tick()
					SelectedValue[NodeData.Name] = NodeData.Stats
					FireDropdownCallback()
				else
					NodeData.LastCB = tick()
					SelectedValue = NodeData.Value
					FireDropdownCallback()
				end
				RefreshVisualNodes()
			end

			local function AddOptionNode(OptionKey, OptionVal)
				local NameStr = tostring((type(OptionKey) ~= "string" or not OptionKey) and OptionVal or OptionKey)
				if not OptionNodes[NameStr] then
					OptionNodes[NameStr] = {
						index = OptionKey,
						Value = OptionVal,
						Name = NameStr,
						Stats = false,
						LastCB = 0,
					}

					if IsMultiSelect then
						local DefaultActive = SelectedValue[NameStr]
						SelectedValue[NameStr] = DefaultActive or false
						OptionNodes[NameStr].Stats = DefaultActive
					end

					local OptionBtn = CreateElement("Button", ListScroll, {
						Name = "Option",
						Size = UDim2.new(1, 0, 0, 21),
						Position = UDim2.new(0, 0, 0.5),
						AnchorPoint = Vector2.new(0, 0.5),
					})
					CreateElement("Corner", OptionBtn, UDim.new(0, 4))

					local IndicatorNode = RegisterInstance(
						CreateInstance("Frame", OptionBtn, {
							Position = UDim2.new(0, 1, 0.5),
							Size = UDim2.new(0, 4, 0, 4),
							BackgroundColor3 = ActiveTheme["Color Theme"],
							BackgroundTransparency = 1,
							AnchorPoint = Vector2.new(0, 0.5),
						}),
						"Theme"
					)
					CreateElement("Corner", IndicatorNode, UDim.new(0.5, 0))

					local TextNode = RegisterInstance(
						CreateInstance("TextLabel", OptionBtn, {
							Size = UDim2.new(1, 0, 1),
							Position = UDim2.new(0, 10),
							Text = NameStr,
							TextColor3 = ActiveTheme["Color Text"],
							Font = Enum.Font.GothamBold,
							TextXAlignment = "Left",
							BackgroundTransparency = 1,
							TextTransparency = 0.4,
						}),
						"Text"
					)

					OptionBtn.Activated:Connect(function()
						SelectNode(OptionNodes[NameStr])
					end)

					OptionNodes[NameStr].nodes = { OptionBtn, IndicatorNode, TextNode }
				end
			end

			local function RemoveOptionNode(OptionKey, OptionVal)
				local NameStr = tostring((type(OptionKey) == "string" and OptionKey) and OptionKey or OptionVal)
				if OptionNodes[NameStr] then
					if IsMultiSelect then
						SelectedValue[NameStr] = nil
					else
						SelectedValue = nil
					end
					OptionNodes[NameStr].nodes[1]:Destroy()
					table.clear(OptionNodes[NameStr])
					OptionNodes[NameStr] = nil
				end
			end

			local function SetDropdownOptions(NewOptions, ClearPrevious)
				if ClearPrevious then
					table.foreach(OptionNodes, RemoveOptionNode)
				end
				table.foreach(NewOptions, AddOptionNode)
				FireDropdownCallback()
				RefreshVisualNodes()
			end

			table.foreach(DropOptions, AddOptionNode)
			FireDropdownCallback()
			RefreshVisualNodes()

			DropInstance.Activated:Connect(ToggleDropdown)
			OverlayButton.MouseButton1Down:Connect(CloseDropdown)
			OverlayButton.MouseButton1Click:Connect(CloseDropdown)
			MainFrame:GetPropertyChangedSignal("Visible"):Connect(CloseDropdown)

			DropButtonBg:GetPropertyChangedSignal("AbsolutePosition"):Connect(AlignDropdownPosition)
			DropInstance.Activated:Connect(RecalculateHeight)
			ListScroll.ChildAdded:Connect(RecalculateHeight)
			ListScroll.ChildRemoved:Connect(RecalculateHeight)

			AlignDropdownPosition()
			RecalculateHeight()

			return {
				Visible = function(_, ...) EventUtils:ToggleVisible(DropInstance, ...) end,
				Destroy = function(_) DropInstance:Destroy() end,
				Callback = function(_, ...) EventUtils:InsertCallback(Callbacks, ...)(SelectedValue) end,
				Add = function(_, ...)
					local Args = {...}
					if type(Args[1]) ~= "table" then
						table.foreach(Args, function(_, Item) AddOptionNode(Item) end)
					else
						table.foreach(Args[1], function(_, Item) AddOptionNode(Item) end)
					end
				end,
				Remove = function(_, Target)
					for Key, NodeData in pairs(OptionNodes) do
						if (type(Target) == "number" and Key == Target) or NodeData.Name == "Option" then
							RemoveOptionNode(Key, NodeData.Value)
						end
					end
				end,
				Select = function(_, Target)
					for Key, NodeData in pairs(OptionNodes) do
						if (type(Target) == "string" and NodeData.Name == Target) or (type(Target) == "number" and Key == Target) then
							SelectNode(NodeData)
						end
					end
				end,
				Set = function(_, NewOptions, KeepCurrent)
					if type(NewOptions) ~= "table" then
						if type(NewOptions) == "function" then Callbacks = NewOptions end
					else
						SetDropdownOptions(NewOptions, not KeepCurrent)
					end
				end,
			}
		end

		function TabFunctions.AddSlider(_, Config)
			local SliderTitle = Config[1] or Config.Name or (Config.Title or "Slider!")
			local SliderDesc = Config.Desc or (Config.Description or "")
			local MinValue = Config[2] or Config.MinValue or (Config.Min or 10)
			local MaxValue = Config[3] or Config.MaxValue or (Config.Max or 100)
			local StepValue = Config[4] or (Config.Increase or 1)
			local Callbacks = EventUtils:GetCallback(Config, 6)
			local SliderFlag = Config[7] or (Config.Flag or false)
			local CurrentValue = Config[5] or (Config.Default or 25)

			if IsFlagSaved(SliderFlag) then
				CurrentValue = GetFlagValue(SliderFlag)
			end

			local BaseMin = MinValue / StepValue
			local BaseMax = MaxValue / StepValue
			local SliderInstance, BaseController = CreateOptionBase(TabContainer, SliderTitle, SliderDesc, UDim2.new(1, -180))

			local InteractionBtn = CreateInstance("TextButton", SliderInstance, {
				Size = UDim2.new(0.45, 0, 1),
				Position = UDim2.new(1),
				AnchorPoint = Vector2.new(1, 0),
				AutoButtonColor = false,
				Text = "",
				BackgroundTransparency = 1,
			})

			local SliderBarBg = RegisterInstance(
				CreateInstance("Frame", InteractionBtn, {
					BackgroundColor3 = ActiveTheme["Color Stroke"],
					Size = UDim2.new(1, -20, 0, 6),
					Position = UDim2.new(0.5, 0, 0.5),
					AnchorPoint = Vector2.new(0.5, 0.5),
				}),
				"Stroke"
			)
			CreateElement("Corner", SliderBarBg)

			local SliderFill = RegisterInstance(
				CreateInstance("Frame", SliderBarBg, {
					BackgroundColor3 = ActiveTheme["Color Theme"],
					Size = UDim2.fromScale(0.3, 1),
					BorderSizePixel = 0,
				}),
				"Theme"
			)
			CreateElement("Corner", SliderFill)

			local SliderKnob = CreateInstance("Frame", SliderBarBg, {
				Size = UDim2.new(0, 6, 0, 12),
				BackgroundColor3 = Color3.fromRGB(220, 220, 220),
				Position = UDim2.fromScale(0.3, 0.5),
				AnchorPoint = Vector2.new(0.5, 0.5),
				BackgroundTransparency = 0.2,
			})
			CreateElement("Corner", SliderKnob)

			local ValueLabel = RegisterInstance(
				CreateInstance("TextLabel", InteractionBtn, {
					Size = UDim2.new(0, 14, 0, 14),
					AnchorPoint = Vector2.new(1, 0.5),
					Position = UDim2.new(0, 0, 0.5),
					BackgroundTransparency = 1,
					TextColor3 = ActiveTheme["Color Text"],
					Font = Enum.Font.FredokaOne,
					TextSize = 12,
				}),
				"Text"
			)

			local LabelScale = CreateInstance("UIScale", ValueLabel)
			local TrackAnchor = CreateInstance("Frame", SliderBarBg, {
				Position = UDim2.new(0, 0, 0.5, 0),
				Visible = false,
			})

			local function FireSliderCallback(Ratio)
				local CalculatedValue = tonumber(Ratio * StepValue)
				local FinalValue = math.floor(CalculatedValue * 100) / 100

				ValueLabel.Text = tostring(FinalValue)
				CurrentValue = FinalValue
				EventUtils:FireCallback(Callbacks, CurrentValue)
			end

			local function UpdateKnobPosition()
				local MouseX = UserInputService:GetMouse().X - TrackAnchor.AbsolutePosition.X
				local RelativeRatio = MouseX / SliderBarBg.AbsoluteSize.X
				SliderKnob.Position = UDim2.new(math.clamp(RelativeRatio, 0, 1), 0, 0.5, 0)
			end

			local function OnSliderMove()
				SliderFill.Size = UDim2.new(SliderKnob.Position.X.Scale, 0, 1, 0)
				local ScaleRatio = SliderKnob.Position.X.Scale
				FireSliderCallback(math.floor(ScaleRatio * BaseMax / BaseMax * (BaseMax - BaseMin) + BaseMin))
			end

			InteractionBtn.MouseButton1Down:Connect(function()
				PlayTween({ SliderKnob, "Transparency", 0, 0.3 })
				TabContainer.ScrollingEnabled = false

				while UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton1) do
					task.wait()
					UpdateKnobPosition()
				end

				PlayTween({ SliderKnob, "Transparency", 0.2, 0.3 })
				TabContainer.ScrollingEnabled = true
				SetFlagValue(SliderFlag, CurrentValue)
			end)

			ValueLabel:GetPropertyChangedSignal("Text"):Connect(function()
				LabelScale.Scale = 0.3
				PlayTween({ LabelScale, "Scale", 1.2, 0.1 })
				PlayTween({ ValueLabel, "Rotation", math.random(-1, 1) * 5, 0.15, true })
				PlayTween({ LabelScale, "Scale", 1, 0.2 })
				PlayTween({ ValueLabel, "Rotation", 0, 0.1 })
			end)

			local function SetSlider(NewValue)
				if type(NewValue) == "number" then
					local ScaledMin = BaseMin * StepValue
					local ScaledMax = BaseMax * StepValue
					local TargetRatio = (NewValue - ScaledMin) / (ScaledMax - ScaledMin)

					SetFlagValue(SliderFlag, NewValue)
					PlayTween({ SliderKnob, "Position", UDim2.fromScale(math.clamp(TargetRatio, 0, 1), 0.5), 0.3, true })
				end
			end

			SetSlider(CurrentValue)
			SliderKnob:GetPropertyChangedSignal("Position"):Connect(OnSliderMove)
			OnSliderMove()

			return {
				Set = function(_, Title, Desc)
					if Title and Desc then
						BaseController:SetTitle(Title)
						BaseController:SetDesc(Desc)
					elseif type(Title) ~= "string" then
						if type(Title) ~= "function" then
							if type(Title) == "number" then SetSlider(Title) end
						else
							Callbacks = Title
						end
					else
						BaseController:SetTitle(Title)
					end
				end,
				Callback = function(_, ...) EventUtils:InsertCallback(Callbacks, ...)(tonumber(CurrentValue)) end,
				Visible = function(_, ...) EventUtils:ToggleVisible(SliderInstance, ...) end,
				Destroy = function(_) SliderInstance:Destroy() end,
			}
		end

		return TabFunctions
	end

	return WindowController
end
